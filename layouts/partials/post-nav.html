{{- /* 上一篇 / 下一篇：按全站 posts 的时间倒序计算，跨目录也连续 */ -}} {{- $all := (where site.RegularPages "Section" "posts").ByDate.Reverse -}} {{- $cur := . -}} {{- $idx := -1 -}} {{- range $i, $p := $all }}{{ if eq $p $cur }}{{ $idx = $i }}{{ end }}{{ end -}} {{- if ge $idx 0 -}} {{- $newer := "" }}{{ if gt $idx 0 }}{{ $newer = index $all (sub $idx 1) }}{{ end -}} {{- $older := "" }}{{ if lt $idx (sub (len $all) 1) }}{{ $older = index $all (add $idx 1) }}{{ end -}}
{{- with $newer }} 较新一篇 {{ .Title }} {{- end }}
{{- with $older }} 较早一篇 {{ .Title }} {{- end }}
{{- end -}}
